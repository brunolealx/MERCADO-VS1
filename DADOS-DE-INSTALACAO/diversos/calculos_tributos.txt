--1. Criar a Tabela de Configuração (Onde moram as regras)

--Antes da View, precisamos da tabela que diz quanto cada CST paga de imposto:
SQL

CREATE TABLE IF NOT EXISTS public.config_tributaria (
    cst_icms VARCHAR(3) PRIMARY KEY,
    aliquota_estimada NUMERIC(5,2),
    descricao_regra TEXT
);

--2. Criar a View vw_pdv_bipagem

--este é o comando que "une" os teus 21.799 produtos com as regras de imposto. É desta View que o teu CaixasForm vai ler os dados:
SQL

CREATE OR REPLACE VIEW public.vw_pdv_bipagem AS
SELECT 
    p.id,
    p.codigo_barra,
    p.descricao,
    p.marca,
    p.preco_venda,
    p.quantidade_estoque,
    p.cst_icms,
    p.preco_custo,
    COALESCE(t.aliquota_estimada, 13.45) as aliquota_aplicada,
    ROUND((p.preco_venda * COALESCE(t.aliquota_estimada, 13.45) / 100), 2) as valor_imposto_item
FROM 
    public.tabela_produtos p
LEFT JOIN 
    public.config_tributaria t ON p.cst_icms = t.cst_icms;
